# Slice 1 — Two-Party Consent Gate Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a deployed Attorney.plus Slice 1 — landing page, async two-party signup-and-consent gate with tamper-evident audit chain, and god admin panel — to a live URL backed by `git@github.com:jeff-cline/attorney.git` auto-deploying on push.

**Architecture:** Next.js 15 App Router monorepo (one app, three audiences: marketing, party dashboard, admin). Drizzle + Neon Postgres with an append-only `agreements` chain (each row hashes the prior row globally). Auth.js v5 with credentials + magic-link, sessions in Postgres. Server actions for all writes; admin role-guarded server-side. Resend for transactional email, fully async — parties never need to be online together. Tailwind + shadcn/ui. Deployed on Vercel; custom domain `attorney.plus`.

**Tech Stack:** Next.js 15 (TS), Drizzle ORM, Neon Postgres, Auth.js v5, bcrypt, Zod, Resend, Tailwind CSS, shadcn/ui, Vitest, Vercel.

```ts
import { db } from "./db";
import { agreements } from "@/db/schema";
import { sha256 } from "./hash";
import { sql, desc } from "drizzle-orm";

export const GENESIS_HASH = sha256("attorney.plus:genesis");

type ChainRow = {
  id: string;
  caseId: string | null;
  userId: string;
  agreementType: "platform_tos" | "arbitration_consent";
  agreementTextHash: string;
  ipAddress: string | null;
  userAgent: string | null;
  createdAt: Date;
  prevHash: string;
};

export function canonicalize(row: ChainRow): string {
  return JSON.stringify({
    id: row.id,
    caseId: row.caseId,
    userId: row.userId,
    agreementType: row.agreementType,
    agreementTextHash: row.agreementTextHash,
    ipAddress: row.ipAddress,
    userAgent: row.userAgent,
    createdAt: row.createdAt.toISOString(),
    prevHash: row.prevHash,
  });
}

export function computeRowHash(row: ChainRow): string {
  return sha256(canonicalize(row));
}

export async function appendAgreement(input: Omit<ChainRow, "id" | "createdAt" | "prevHash">) {
  return db.transaction(async (tx) => {
    const [tail] = await tx.select({ rowHash: agreements.rowHash })
      .from(agreements)
      .orderBy(desc(agreements.seq))
      .limit(1);
    const prevHash = tail?.rowHash ?? GENESIS_HASH;
    const id = crypto.randomUUID();
    const createdAt = new Date();
    const rowHash = computeRowHash({ ...input, id, createdAt, prevHash });
    await tx.insert(agreements).values({
      id, caseId: input.caseId, userId: input.userId,
      agreementType: input.agreementType,
      agreementTextHash: input.agreementTextHash,
      ipAddress: input.ipAddress, userAgent: input.userAgent,
      prevHash, rowHash, createdAt,
    });
    return { id, rowHash, prevHash, createdAt };
  });
}

export async function verifyChain(): Promise<{ ok: boolean; brokenAtId?: string; checked: number }> {
  const rows = await db.select().from(agreements).orderBy(agreements.seq);
  let prev = GENESIS_HASH;
  for (const r of rows) {
    if (r.prevHash !== prev) return { ok: false, brokenAtId: r.id, checked: rows.length };
    const expected = computeRowHash({
      id: r.id, caseId: r.caseId, userId: r.userId,
      agreementType: r.agreementType,
      agreementTextHash: r.agreementTextHash,
      ipAddress: r.ipAddress, userAgent: r.userAgent,
      createdAt: r.createdAt, prevHash: r.prevHash,
    });
    if (expected !== r.rowHash) return { ok: false, brokenAtId: r.id, checked: rows.length };
    prev = r.rowHash;
  }
  return { ok: true, checked: rows.length };
}
```

const ALPHABET = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // 32 chars, no 0/O/1/I
export const INVITE_CODE_REGEX = /^ATTPLUS-[ABCDEFGHJKLMNPQRSTUVWXYZ23456789]{6}$/;

export function generateInviteCode(): string {
  let s = "";
  const buf = new Uint32Array(6);
  crypto.getRandomValues(buf);
  for (const v of buf) s += ALPHABET[v % ALPHABET.length];
  return `ATTPLUS-${s}`;
}

import bcrypt from "bcryptjs";
import { z } from "zod";
import { db } from "./db";
import { users } from "@/db/schema";
import { eq } from "drizzle-orm";
import { env } from "./env";

const credSchema = z.object({ email: z.string().email(), password: z.string().min(1) });

export const { handlers, auth, signIn, signOut } = NextAuth({
  adapter: DrizzleAdapter(db),
  session: { strategy: "jwt" },
  secret: env.AUTH_SECRET,
  pages: { signIn: "/auth/login" },
  providers: [
    Credentials({
      name: "Email + Password",
      credentials: { email: {}, password: {} },
      async authorize(creds) {
        const parsed = credSchema.safeParse(creds);
        if (!parsed.success) return null;
        const u = await db.query.users.findFirst({
          where: eq(users.email, parsed.data.email.toLowerCase()),
        });
        if (!u || !u.passwordHash) return null;
        const ok = await bcrypt.compare(parsed.data.password, u.passwordHash);
        if (!ok) return null;
        return { id: u.id, email: u.email, name: u.displayName ?? null, role: u.role };
      },
    }),
    ...(env.RESEND_API_KEY ? [Resend({
      apiKey: env.RESEND_API_KEY,
      from: env.RESEND_FROM,
    })] : []),
  ],
  callbacks: {
    async jwt({ token, user }) {
      if (user) {
        // @ts-expect-error - user.role is set by authorize
        token.role = user.role;
        token.uid = user.id;
      }
      return token;
    },
    async session({ session, token }) {
      // @ts-expect-error - augmenting
      session.user.role = token.role;
      // @ts-expect-error
      session.user.id = token.uid;
      return session;
    },
  },
});

export async function requireAdmin() {
  const s = await auth();
  // @ts-expect-error
  if (!s?.user || s.user.role !== "admin") throw new Error("forbidden");
  return s;
}

"use server";
import { z } from "zod";
import { db } from "@/lib/db";
import { cases } from "@/db/schema";
import { generateInviteCode } from "@/lib/codes";
import { auth } from "@/lib/auth";
import { eq, and } from "drizzle-orm";
import { appendAgreement } from "@/lib/audit";
import { sha256 } from "@/lib/hash";
import { captureRequestMeta } from "@/lib/ip";
import { currentTos } from "@/lib/tos";

const ARBITRATION_TEXT = "I agree to use Attorney.plus arbitration as defined in the current Terms.";

export async function createCase() {
  const session = await auth();
  // @ts-expect-error
  if (!session?.user?.id) throw new Error("unauthenticated");
  // @ts-expect-error
  const userId = session.user.id as string;

  // generate unique invite code (retry up to 5 times)
  for (let i = 0; i < 5; i++) {
    const code = generateInviteCode();
    try {
      const [c] = await db.insert(cases).values({
        inviteCode: code,
        initiatorId: userId,
      }).returning();
      return c;
    } catch (e: any) {
      if (!String(e).includes("cases_invite_code_uq")) throw e;
    }
  }
  throw new Error("could not generate unique code");
}

const JoinInput = z.object({
  inviteCode: z.string().regex(/^ATTPLUS-[A-Z2-9]{6}$/),
});

export async function joinCase(fd: FormData) {
  const session = await auth();
  // @ts-expect-error
  if (!session?.user?.id) throw new Error("unauthenticated");
  // @ts-expect-error
  const userId = session.user.id as string;
  const { inviteCode } = JoinInput.parse(Object.fromEntries(fd));

  const c = await db.query.cases.findFirst({ where: eq(cases.inviteCode, inviteCode) });
  if (!c) throw new Error("invalid code");
  if (c.initiatorId === userId) throw new Error("cannot join your own case");
  if (c.joinerId && c.joinerId !== userId) throw new Error("case already joined");

  const [updated] = await db.update(cases).set({
    joinerId: userId,
    status: "pending_agreements",
    updatedAt: new Date(),
  }).where(eq(cases.id, c.id)).returning();
  return updated;
}

export async function agreeToArbitration(caseId: string) {
  const session = await auth();
  // @ts-expect-error
  if (!session?.user?.id) throw new Error("unauthenticated");
  // @ts-expect-error
  const userId = session.user.id as string;
  const meta = await captureRequestMeta();
  const tos = await currentTos();

  const c = await db.query.cases.findFirst({ where: eq(cases.id, caseId) });
  if (!c) throw new Error("case not found");
  const isInitiator = c.initiatorId === userId;
  const isJoiner = c.joinerId === userId;
  if (!isInitiator && !isJoiner) throw new Error("not a party to this case");

  await appendAgreement({
    caseId: c.id,
    userId,
    agreementType: "arbitration_consent",
    agreementTextHash: sha256(`${ARBITRATION_TEXT}::${tos.version}`),
    ipAddress: meta.ip,
    userAgent: meta.userAgent,
  });

  const patch: Partial<typeof cases.$inferInsert> = { updatedAt: new Date() };
  if (isInitiator && !c.initiatorAgreedAt) patch.initiatorAgreedAt = new Date();
  if (isJoiner && !c.joinerAgreedAt) patch.joinerAgreedAt = new Date();

  const both =
    (patch.initiatorAgreedAt ?? c.initiatorAgreedAt) &&
    (patch.joinerAgreedAt ?? c.joinerAgreedAt);
  if (both) patch.status = "ready_for_intake";

  const [updated] = await db.update(cases).set(patch).where(eq(cases.id, c.id)).returning();
  return updated;
}

```ts
import { Resend } from "resend";
import { db } from "./db";
import { emailLog } from "@/db/schema";
import { env } from "./env";

const client = env.RESEND_API_KEY ? new Resend(env.RESEND_API_KEY) : null;

export async function sendTemplated(params: {
  to: string;
  subject: string;
  html: string;
  template: string;
  payload: object;
}) {
  if (!client) {
    await db.insert(emailLog).values({
      toEmail: params.to,
      template: params.template,
      payload: JSON.stringify(params.payload),
      error: "RESEND_API_KEY not set; skipped",
    });
    return;
  }
  try {
    const r = await client.emails.send({
      from: env.RESEND_FROM,
      to: params.to,
      subject: params.subject,
      html: params.html,
    });
    await db.insert(emailLog).values({
      toEmail: params.to,
      template: params.template,
      payload: JSON.stringify(params.payload),
      resendMessageId: r.data?.id,
    });
  } catch (e: any) {
    await db.insert(emailLog).values({
      toEmail: params.to,
      template: params.template,
      payload: JSON.stringify(params.payload),
      error: String(e?.message ?? e),
    });
  }
}
```
